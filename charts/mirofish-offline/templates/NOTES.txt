--------------------------------------------------------------------------
{{- if .Values.ollama.modelPull.enabled }}
The chart is downloading {{ .Values.ollama.models.chat }} and {{ .Values.ollama.models.embedding }} into

simulation will fail with model-not-found errors.

  kubectl logs -n {{ .Release.Namespace }} -f job/{{ include "mirofish.modelPullJobName" . }}

  kubectl exec -n {{ .Release.Namespace }} sts/{{ include "mirofish.ollama.fullname" . }} -- ollama pull {{ .Values.ollama.models.chat }}
  kubectl exec -n {{ .Release.Namespace }} sts/{{ include "mirofish.ollama.fullname" . }} -- ollama pull {{ .Values.ollama.models.embedding }}
{{- end }}
